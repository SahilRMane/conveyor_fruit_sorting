cmake_minimum_required(VERSION 3.16)
project(sorting_supervisor)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(mujoco_vendor REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(yaml_cpp_vendor REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(OpenGL REQUIRED COMPONENTS OpenGL EGL)
find_package(moveit_core REQUIRED)
find_package(rclcpp REQUIRED)
find_package(moveit_msgs REQUIRED)
# These project headers implement the existing node-free physical checks.
add_executable(render_m4a_views src/render_m4a_views.cpp)
target_include_directories(render_m4a_views PRIVATE ../sorting_simulation/include)
target_link_libraries(render_m4a_views mujoco::mujoco Eigen3::Eigen yaml-cpp OpenGL::OpenGL OpenGL::EGL)
target_compile_options(render_m4a_views PRIVATE -Wall -Wextra)
install(TARGETS render_m4a_views DESTINATION lib/${PROJECT_NAME})
add_executable(render_m4b_dataset src/render_m4b_dataset.cpp)
target_include_directories(render_m4b_dataset PRIVATE ../sorting_simulation/include)
target_link_libraries(render_m4b_dataset mujoco::mujoco Eigen3::Eigen yaml-cpp OpenGL::OpenGL OpenGL::EGL)
target_compile_options(render_m4b_dataset PRIVATE -Wall -Wextra)
install(TARGETS render_m4b_dataset DESTINATION lib/${PROJECT_NAME})
add_executable(verify_m4a_scene src/verify_m4a_scene.cpp)
ament_target_dependencies(verify_m4a_scene moveit_core rclcpp moveit_msgs)
target_link_libraries(verify_m4a_scene yaml-cpp)
target_compile_options(verify_m4a_scene PRIVATE -Wall -Wextra)
install(TARGETS verify_m4a_scene DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/validate_m4a.py scripts/test_m4a.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/inspection_core.py scripts/m4a_common.py scripts/m4a_runtime.py DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/train_m4b.py scripts/validate_m4b.py scripts/test_m4b.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/m4b_common.py scripts/m4b_data.py scripts/m4b_core.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY models DESTINATION share/${PROJECT_NAME})
add_executable(probe_m4c_scene src/probe_m4c_scene.cpp)
target_include_directories(probe_m4c_scene PRIVATE ../sorting_simulation/include)
target_link_libraries(probe_m4c_scene mujoco::mujoco Eigen3::Eigen yaml-cpp OpenGL::OpenGL OpenGL::EGL)
target_compile_options(probe_m4c_scene PRIVATE -Wall -Wextra)
install(TARGETS probe_m4c_scene DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/m4c_common.py scripts/m4c_core.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/m4c_runtime.py scripts/m4c_evaluator.py scripts/m4c_checks.py scripts/m4c_timing.py scripts/m4c_upstream.py DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/test_m4c.py scripts/validate_m4c.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/m4d_core.py scripts/m4d_handoff.py scripts/m4d_common.py scripts/m4d_pickup.py scripts/m4d_runtime.py scripts/m4d_evaluator.py DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/validate_m4d.py scripts/test_m4d.py scripts/m4d_filter.py DESTINATION lib/${PROJECT_NAME})
ament_package()
